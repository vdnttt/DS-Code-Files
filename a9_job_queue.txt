#include <stdio.h>
#define SIZE 5

int q[SIZE], front = 0, rear = -1;

int main() {
    int ch, job, i;
    do {
        printf("1.Add\n2.Delete\n3.Display\n0.Exit\n");
        scanf("%d", &ch);

        switch (ch) {
            case 1:
                if (rear == SIZE - 1)
                    printf("Queue Overflow\n");
                else {
                    printf("Enter job ID: ");
                    scanf("%d", &job);
                    q[++rear] = job;
                }
                break;

            case 2:
                if (front > rear)
                    printf("Queue Underflow\n");
                else
                    printf("Deleted: %d\n", q[front++]);
                break;

            case 3:
                if (front > rear)
                    printf("Queue is empty\n");
                else {
                    for (i = front; i <= rear; i++)
                        printf("%d ", q[i]);
                    printf("\n");
                }
                break;
        }
    } while (ch != 0);

    return 0;
}

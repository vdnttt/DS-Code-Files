#include <stdio.h>

int main() {
    int a[10][10], b[10][10], r[10][10];
    int r1, c1, r2, c2, i, j, k;

    printf("enter rows and columns of A: ");
    scanf("%d%d", &r1, &c1);

    printf("Enter A:\n");
    for (i = 0; i < r1; i++)
        for (j = 0; j < c1; j++)
            scanf("%d", &a[i][j]);

    printf("ENter rows and columns of B: ");
    scanf("%d%d", &r2, &c2);

    printf("Enter B:\n");
    for (i = 0; i < r2; i++)
        for (j = 0; j < c2; j++)
            scanf("%d", &b[i][j]);

    if (r1 == r2 && c1 == c2) {
        printf("ADdition:\n");
        for (i = 0; i < r1; i++) {
            for (j = 0; j < c1; j++)
                printf("%d ", a[i][j] + b[i][j]);
            printf("\n");
        }

        printf("Subtrction:\n");
        for (i = 0; i < r1; i++) {
            for (j = 0; j < c1; j++)
                printf("%d ", a[i][j] - b[i][j]);
            printf("\n");
        }
    } else
        printf("Addition/subtraction not possible\n");

    if (c1 == r2) {
        printf("Multiplication:\n");
        for (i = 0; i < r1; i++) {
            for (j = 0; j < c2; j++) {
                r[i][j] = 0;
                for (k = 0; k < c1; k++)
                    r[i][j] += a[i][k] * b[k][j];
                printf("%d ", r[i][j]);
            }
            printf("\n");
        }
    } else
        printf("Multiplication not possible\n");

    printf("transpose of A:\n");
    for (i = 0; i < c1; i++) {
        for (j = 0; j < r1; j++)
            printf("%d ", a[j][i]);
        printf("\n");
    }

    return 0;
}
